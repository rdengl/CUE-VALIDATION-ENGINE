Request: {
    // Integer range with min/max value validation
    age: int & >=18 & <=60 @tag(message="Age must be between 18 and 60", min_chars="1")

    // String with required substring, min/max length
    status: string & != "" @tag(must_contain="active", min_chars="6", max_chars="20", message="Status must contain 'active' and be 6-20 chars")

    // String must NOT contain certain text
    username: string & != "" @tag(must_not_contain="admin", message="Username must not contain 'admin'")

    // Decimal precision fixed
    price_exact: number & > 0 @tag(decimal="2", message="Price must have exactly 2 decimal places")

    // Decimal precision max allowed
    price_max: number & > 0 @tag(decimal_max="3", message="Price must have at most 3 decimal places")

    // Date format validation
    start_date: string @tag(date="yyyy-MM-dd", message="Date must be in yyyy-MM-dd format")

    // List validations: must contain all required values
    required_items: [...string] @tag(required_all="apple,banana", message="Must include apple and banana")

    // List validations: must contain at least one of the given values
    any_items: [...string] @tag(contains_any="red,blue,green", message="Must contain at least one of red, blue, green")

    // List validations: must NOT contain any of the given values
    no_items: [...string] @tag(not_contains="forbidden1,forbidden2", message="Must not contain forbidden items")

    // Nested object validation
    address: {
        street: string & != "" @tag(min_chars="5", max_chars="50", message="Street name must be between 5-50 characters")
        zip: string & =~"^[0-9]{5}$" @tag(message="ZIP must be exactly 5 digits")
    }

    // Array of nested objects with validations
    contacts: [...{
        name: string & != "" @tag(message="Contact name must not be empty")
        phone: string & =~"^[0-9]{10}$" @tag(message="Phone must be 10 digits")
    }]

    // Combination: parent rule + child structure
    projects: [...{
        id: string & != "" @tag(message="Project ID required")
        tags: [...string] @tag(contains_any="urgent,important", message="Tags must contain 'urgent' or 'important'")
    }]
}
